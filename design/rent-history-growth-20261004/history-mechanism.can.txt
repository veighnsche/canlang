contract ResourceEvidence {active:bool,capacity:int,pooled:bool,timezone:timezone}
history Resource as ResourceEvidence until=null read=reservation_manager and can_work(actor,row.location)
history Booking as ReportBooking until=null read=reservation_manager and can_work(actor,row.parent.location)
history VenueReservation as ReportVenue until=null read=reservation_manager and can_work(actor,row.parent.location)
history Window as ReportWindow until=null read=reservation_manager and can_work(actor,row.parent.location)
history Downtime as ReportDowntime until=null read=reservation_manager and can_work(actor,row.parent.location)
  export scenario resource_report(resource:Resource,from:datetime,until:datetime) read=true scope=authority -> ReportBatch by=reservation_manager label="Workspace source measures"@{nl="Werkplekbronmaatstaven"}
   require can_work(actor,resource.location) and from<until
   do
    let days=dates(add_days(local_date(from,resource.timezone),-1),add_days(local_date(until,resource.timezone),1),1000)
    let openings=resource.location.LocationPolicy
    let weekly=flatten(flatten(openings as policy select flatten(days as day select (policy.weekly as hours where weekday(day)==hours.weekday select [local_instant(day,hours.opens,policy.timezone,fold=hours.fold),local_instant(add_days(day,hours.close_after),hours.closes,policy.timezone,fold=hours.fold)]))))
    let dated=flatten(flatten(openings as policy select (policy.dated as hours where not hours.closed select [local_instant(hours.day,hours.opens,policy.timezone,fold=hours.fold),local_instant(add_days(hours.day,hours.close_after),hours.closes,policy.timezone,fold=hours.fold)])))
    let midnight=days as day select local_instant(day,"00:00",resource.timezone,fold=earlier)
    let resources=history(resource,from,until)
    let resources_states=resources.entries+(resources.current as item where now<until select {key=item.key,from=max([from,now]),until,sequence=resources.current_sequence,value=item.value})
    let resources_coverage=resources.coverage+([now] as point where point<until select {from=max([from,point]),until,sequence=resources.current_sequence})
    let bookings=history(resource.Booking,from,until)
    let bookings_states=bookings.entries+(bookings.current as item where now<until select {key=item.key,from=max([from,now]),until,sequence=bookings.current_sequence,value=item.value})
    let bookings_coverage=bookings.coverage+([now] as point where point<until select {from=max([from,point]),until,sequence=bookings.current_sequence})
    let venues=history(resource.VenueReservation,from,until)
    let venues_states=venues.entries+(venues.current as item where now<until select {key=item.key,from=max([from,now]),until,sequence=venues.current_sequence,value=item.value})
    let venues_coverage=venues.coverage+([now] as point where point<until select {from=max([from,point]),until,sequence=venues.current_sequence})
    let windows=history(resource.Window,from,until)
    let windows_states=windows.entries+(windows.current as item where now<until select {key=item.key,from=max([from,now]),until,sequence=windows.current_sequence,value=item.value})
    let windows_coverage=windows.coverage+([now] as point where point<until select {from=max([from,point]),until,sequence=windows.current_sequence})
    let downtime=history(resource.Downtime,from,until)
    let downtime_states=downtime.entries+(downtime.current as item where now<until select {key=item.key,from=max([from,now]),until,sequence=downtime.current_sequence,value=item.value})
    let downtime_coverage=downtime.coverage+([now] as point where point<until select {from=max([from,point]),until,sequence=downtime.current_sequence})
    let calendar_edges=flatten(windows_states as entry select [entry.value.from,entry.value.until])+flatten(downtime_states as entry select [entry.value.from,entry.value.until??until])
    let service_edges=flatten(bookings_states as entry select [entry.value.from,entry.value.until,entry.value.checked_in??from,entry.value.checked_out??now])+flatten(bookings_states as entry select flatten(entry.value.intervals as interval select [interval.from+entry.value.arrival_buffer,interval.until-entry.value.departure_buffer]))
    let venue_edges=flatten(venues_states as entry select [entry.value.from,entry.value.until])
    let resources_edges=flatten(resources_states as entry select [entry.from,entry.until])+flatten(resources_coverage as span select [span.from,span.until])
    let bookings_edges=flatten(bookings_states as entry select [entry.from,entry.until])+flatten(bookings_coverage as span select [span.from,span.until])
    let venues_edges=flatten(venues_states as entry select [entry.from,entry.until])+flatten(venues_coverage as span select [span.from,span.until])
    let windows_edges=flatten(windows_states as entry select [entry.from,entry.until])+flatten(windows_coverage as span select [span.from,span.until])
    let downtime_edges=flatten(downtime_states as entry select [entry.from,entry.until])+flatten(downtime_coverage as span select [span.from,span.until])
    let boundaries=[from,until,now]+weekly+dated+midnight+calendar_edges+service_edges+venue_edges+resources_edges+bookings_edges+venues_edges+windows_edges+downtime_edges+(openings as policy select policy.effective)
    let points=group((boundaries as point where from<=point and point<=until) as point,point) as point order=point.key select point.key
    let rows=flatten(points as left where left<until and any(resources_coverage as span,span.from<=left and next_boundary(points,left,until)<=span.until) and any(bookings_coverage as span,span.from<=left and next_boundary(points,left,until)<=span.until) and any(venues_coverage as span,span.from<=left and next_boundary(points,left,until)<=span.until) and any(windows_coverage as span,span.from<=left and next_boundary(points,left,until)<=span.until) and any(downtime_coverage as span,span.from<=left and next_boundary(points,left,until)<=span.until) select flatten(resources_states as policy where policy.from<=left and left<policy.until select flatten(resource.location.LocationPolicy as opening where opening==first(resource.location.LocationPolicy as candidate where candidate.effective<=left order=-candidate.sequence) select [Contribution {source=resource.id,revision=max((resources_coverage as span where span.from<=left and left<span.until select span.sequence)+(bookings_coverage as span where span.from<=left and left<span.until select span.sequence)+(venues_coverage as span where span.from<=left and left<span.until select span.sequence)+(windows_coverage as span where span.from<=left and left<span.until select span.sequence)+(downtime_coverage as span where span.from<=left and left<span.until select span.sequence))??policy.sequence,location=resource.location.id,from=left,until=next_boundary(points,left,until),metric=booked_minutes,provisional=left>=now,quantity=((next_boundary(points,left,until)-left)/1m)*sum((bookings_states as entry where entry.from<=left and left<entry.until select entry.value) as booking where booking.status in [confirmed,occupied,completed] and saved_service(booking,left) select booking.quantity)+((next_boundary(points,left,until)-left)/1m)*sum((venues_states as entry where entry.from<=left and left<entry.until select entry.value) as venue where venue.status==confirmed and venue.from<=left and left<venue.until select venue.quantity),unit=first(["seat-minute"] as unit where policy.value.pooled) ?? "room-minute"},Contribution {source=resource.id,revision=max((resources_coverage as span where span.from<=left and left<span.until select span.sequence)+(bookings_coverage as span where span.from<=left and left<span.until select span.sequence)+(venues_coverage as span where span.from<=left and left<span.until select span.sequence)+(windows_coverage as span where span.from<=left and left<span.until select span.sequence)+(downtime_coverage as span where span.from<=left and left<span.until select span.sequence))??policy.sequence,location=resource.location.id,from=left,until=next_boundary(points,left,until),metric=occupied_minutes,quantity=((next_boundary(points,left,until)-left)/1m)*sum((bookings_states as entry where entry.from<=left and left<entry.until select entry.value) as booking where booking.checked_in!=null and booking.checked_in<=left and left<(booking.checked_out ?? now) select booking.quantity),unit=first(["seat-minute"] as unit where policy.value.pooled) ?? "room-minute",provisional=left>=now or any((bookings_states as entry where entry.from<=left and left<entry.until select entry.value) as booking,booking.checked_in!=null and booking.checked_out==null)},Contribution {source=resource.id,revision=max((resources_coverage as span where span.from<=left and left<span.until select span.sequence)+(bookings_coverage as span where span.from<=left and left<span.until select span.sequence)+(venues_coverage as span where span.from<=left and left<span.until select span.sequence)+(windows_coverage as span where span.from<=left and left<span.until select span.sequence)+(downtime_coverage as span where span.from<=left and left<span.until select span.sequence))??policy.sequence,location=resource.location.id,from=left,until=next_boundary(points,left,until),metric=saleable_minutes,provisional=left>=now,quantity=((next_boundary(points,left,until)-left)/1m)*sum([policy.value.capacity] as capacity where (policy.value.active and policy_open(opening,left,next_boundary(points,left,until)) and any(windows_states as entry,entry.from<=left and left<entry.until and not entry.value.closed and entry.value.from<=left and next_boundary(points,left,until)<=entry.value.until) and not any(windows_states as entry,entry.from<=left and left<entry.until and entry.value.closed and overlaps(left,next_boundary(points,left,until),entry.value.from,entry.value.until)) and not any(downtime_states as entry,entry.from<=left and left<entry.until and entry.value.active and entry.value.from<next_boundary(points,left,until) and (entry.value.until==null or left<entry.value.until))) select capacity),unit=first(["seat-minute"] as unit where policy.value.pooled) ?? "room-minute"},Contribution {source=resource.id,revision=max((resources_coverage as span where span.from<=left and left<span.until select span.sequence)+(bookings_coverage as span where span.from<=left and left<span.until select span.sequence)+(venues_coverage as span where span.from<=left and left<span.until select span.sequence)+(windows_coverage as span where span.from<=left and left<span.until select span.sequence)+(downtime_coverage as span where span.from<=left and left<span.until select span.sequence))??policy.sequence,location=resource.location.id,from=left,until=next_boundary(points,left,until),metric=excluded_minutes,provisional=left>=now,quantity=((next_boundary(points,left,until)-left)/1m)*sum([policy.value.capacity] as capacity where not (policy.value.active and policy_open(opening,left,next_boundary(points,left,until)) and any(windows_states as entry,entry.from<=left and left<entry.until and not entry.value.closed and entry.value.from<=left and next_boundary(points,left,until)<=entry.value.until) and not any(windows_states as entry,entry.from<=left and left<entry.until and entry.value.closed and overlaps(left,next_boundary(points,left,until),entry.value.from,entry.value.until)) and not any(downtime_states as entry,entry.from<=left and left<entry.until and entry.value.active and entry.value.from<next_boundary(points,left,until) and (entry.value.until==null or left<entry.value.until))) select capacity),unit=first(["seat-minute"] as unit where policy.value.pooled) ?? "room-minute"}])))
    let complete=sum(resources.coverage as span select span.until-span.from)==max([min([until,now])-from,0s]) and sum(bookings.coverage as span select span.until-span.from)==max([min([until,now])-from,0s]) and sum(venues.coverage as span select span.until-span.from)==max([min([until,now])-from,0s]) and sum(windows.coverage as span select span.until-span.from)==max([min([until,now])-from,0s]) and sum(downtime.coverage as span select span.until-span.from)==max([min([until,now])-from,0s]) and any(openings as policy,policy.effective<=from)
    let stamp=join([resources.revision,bookings.revision,venues.revision,windows.revision,downtime.revision],":")+":"+join((openings as policy order=policy.sequence select format("{id}:{sequence}",{id=policy.id,sequence=policy.sequence})),",")
    if complete
     return ReportBatch {checkpoint=Checkpoint {source=resource.id,revision=stamp,generated=now,state=fresh,complete=true},rows}
    return ReportBatch {checkpoint=Checkpoint {source=resource.id,revision=stamp,generated=now,state=partial,complete=false},rows}
