  derive saved_saleable(policy:ResourcePolicy,opening:LocationPolicy,left:datetime,right:datetime):bool = policy.value.active and policy_open(opening,left,right) and any(policy.value.windows as window,not window.closed and window.from<=left and right<=window.until) and not any(policy.value.windows as window,window.closed and overlaps(left,right,window.from,window.until)) and not any(policy.value.downtime as downtime,downtime.active and downtime.from<right and (downtime.until==null or left<downtime.until))
  export scenario resource_report(resource:Resource,from:datetime,until:datetime) read=true scope=authority -> ReportBatch by=reservation_manager label="Workspace source measures"@{nl="Werkplekbronmaatstaven"}
   require can_work(actor,resource.location) and from<until
   do
    let days=dates(add_days(local_date(from,resource.timezone),-1),add_days(local_date(until,resource.timezone),1),1000)
    let policies=resource.ResourcePolicy
    let openings=resource.location.LocationPolicy
    let weekly=flatten(flatten(openings as policy select flatten(days as day select (policy.weekly as hours where weekday(day)==hours.weekday select [local_instant(day,hours.opens,policy.timezone,fold=hours.fold),local_instant(add_days(day,hours.close_after),hours.closes,policy.timezone,fold=hours.fold)]))))
    let dated=flatten(flatten(openings as policy select (policy.dated as hours where not hours.closed select [local_instant(hours.day,hours.opens,policy.timezone,fold=hours.fold),local_instant(add_days(hours.day,hours.close_after),hours.closes,policy.timezone,fold=hours.fold)])))
    let midnight=days as day select local_instant(day,"00:00",resource.timezone,fold=earlier)
    let windows=flatten(flatten(policies as policy select (policy.value.windows as window select [window.from,window.until])))
    let downtime=flatten(flatten(policies as policy select (policy.value.downtime as downtime select [downtime.from,downtime.until ?? until])))
    let service=flatten(flatten(policies as policy select (policy.value.bookings as booking select [booking.from,booking.until,booking.checked_in ?? from,booking.checked_out ?? now])))
    let access=flatten(flatten(policies as policy select (policy.value.bookings as booking select flatten(booking.intervals as interval select [interval.from+booking.arrival_buffer,interval.until-booking.departure_buffer]))))
    let venues=flatten(flatten(policies as policy select (policy.value.venues as venue select [venue.from,venue.until])))
    let boundaries=[from,until]+weekly+dated+midnight+windows+downtime+service+access+venues+(policies as policy select policy.effective)+(openings as policy select policy.effective)
    let points=group((boundaries as point where from<=point and point<=until) as point,point) as point order=point.key select point.key
    let complete=any(policies as policy,policy.effective<=from) and any(openings as policy,policy.effective<=from)
    let stamp=join((policies as policy order=policy.sequence select format("{id}:{sequence}",{id=policy.id,sequence=policy.sequence})),",")+":"+join((openings as policy order=policy.sequence select format("{id}:{sequence}",{id=policy.id,sequence=policy.sequence})),",")
    if complete
     return ReportBatch {checkpoint=Checkpoint {source=resource.id,revision=stamp,generated=now,state=fresh,complete=true},rows=saved_rows(resource,points,until)}
    return ReportBatch {checkpoint=Checkpoint {source=resource.id,revision=stamp,generated=now,state=partial,complete=false},rows=saved_rows(resource,points,until)}
